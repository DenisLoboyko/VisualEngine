const char* vertSrc = R"(
#version 330 core
layout(location=0) in vec3 aPos;
layout(location=1) in vec3 aNormal;
out vec3 FragPos; out vec3 Normal;
uniform mat4 model,view,projection;
void main(){
    FragPos=vec3(model*vec4(aPos,1.0));
    Normal=mat3(transpose(inverse(model)))*aNormal;
    gl_Position=projection*view*vec4(FragPos,1.0);
})";

const char* fragSrc = R"(
#version 330 core
in vec3 FragPos,Normal;
out vec4 FragColor;
uniform vec3 objectColor,lightPos,lightColor,viewPos;
void main(){
    vec3 norm=normalize(Normal);
    vec3 ld=normalize(lightPos-FragPos);
    float diff=max(dot(norm,ld),0.0);
    vec3 vd=normalize(viewPos-FragPos);
    float spec=pow(max(dot(vd,reflect(-ld,norm)),0.0),32);
    FragColor=vec4((0.15+diff*0.8+spec*0.3)*objectColor*lightColor,1.0);
})";

const char* outlineVert = R"(
#version 330 core
layout(location=0) in vec3 aPos;
layout(location=1) in vec3 aNormal;
uniform mat4 model,view,projection;
uniform float outlineSize;
void main(){
    gl_Position=projection*view*model*vec4(aPos+aNormal*outlineSize,1.0);
})";

const char* outlineFrag = R"(
#version 330 core
out vec4 FragColor;
uniform vec4 outlineColor;
void main(){ FragColor=outlineColor; })";

enum class PrimitiveType { Cube, Sphere, Cylinder, Pyramid, Model3D };

struct SceneObject {
    std::string name;

    PrimitiveType type;
    std::string modelPath;

struct Ray { glm::vec3 origin, dir; };

Ray screenToRay(double mx, double my, int w, int h, const glm::mat4& view, const glm::mat4& proj)
{
    float nx=(2.0f*mx/w)-1.0f, ny=1.0f-(2.0f*my/h);
    glm::vec4 eye=glm::inverse(proj)*glm::vec4(nx,ny,-1,1);
    eye=glm::vec4(eye.x,eye.y,-1,0);
    return { glm::vec3(glm::inverse(view)*glm::vec4(0,0,0,1)),
             glm::normalize(glm::vec3(glm::inverse(view)*eye)) };
}

bool rayAABB(const Ray& r, glm::vec3 c, glm::vec3 hs, float& t)
{
    glm::vec3 mn=c-hs, mx=c+hs;
    float tmin=-1e9f, tmax=1e9f;
    for(int i=0;i<3;i++){
        if(fabs(r.dir[i])<1e-6f){ if(r.origin[i]<mn[i]||r.origin[i]>mx[i]) return false; }
        else{
            float t1=(mn[i]-r.origin[i])/r.dir[i], t2=(mx[i]-r.origin[i])/r.dir[i];
            if(t1>t2) std::swap(t1,t2);
            tmin=std::max(tmin,t1); tmax=std::min(tmax,t2);
            if(tmin>tmax) return false;
        }
    }
    t=tmin>0?tmin:tmax; return t>0;
}

float rayPlaneT(const Ray& r, glm::vec3 n, glm::vec3 p)
{
    float d=glm::dot(n,r.dir);
    if(fabs(d)<1e-6f) return -1;
    return glm::dot(n,p-r.origin)/d;
}

bool gizmoArrowHit(const Ray& ray, glm::vec3 objPos, glm::vec3 axisDir, float gs, float& t)
{
    glm::vec3 end=objPos+axisDir*gs;
    glm::vec3 mn=glm::min(objPos,end)-0.08f*gs;
    glm::vec3 mx=glm::max(objPos,end)+0.08f*gs;
    return rayAABB(ray,(mn+mx)*0.5f,(mx-mn)*0.5f,t);
}

GizmoAxis dragAxis=GizmoAxis::None;
glm::vec3 dragStartPos,dragStartRot,dragStartScale,dragStartHit;

void mouse_callback(GLFWwindow* w, double x, double y)
{
    mouseX=x; mouseY=y;
    if(!rightMouseDown) return;
    if(firstMouse){lastX=x;lastY=y;firstMouse=false;}
    camera.ProcessMouse(x-lastX,lastY-y);
    lastX=x; lastY=y;
}

void mouse_button_callback(GLFWwindow* w, int btn, int action, int mods)
{
    if(btn==GLFW_MOUSE_BUTTON_RIGHT){rightMouseDown=(action==GLFW_PRESS);firstMouse=true;}
    if(btn==GLFW_MOUSE_BUTTON_LEFT){
        if(action==GLFW_PRESS){leftDown=true;leftClickThisFrame=true;glfwGetCursorPos(w,&clickX,&clickY);}
        else{leftDown=false;dragAxis=GizmoAxis::None;}
    }
}

    float v[]={
        -0.5f,-0.5f,-0.5f,0,0,-1,0.5f,-0.5f,-0.5f,0,0,-1,0.5f,0.5f,-0.5f,0,0,-1,-0.5f,0.5f,-0.5f,0,0,-1,
        -0.5f,-0.5f,0.5f,0,0,1,0.5f,-0.5f,0.5f,0,0,1,0.5f,0.5f,0.5f,0,0,1,-0.5f,0.5f,0.5f,0,0,1,
        -0.5f,-0.5f,-0.5f,-1,0,0,-0.5f,0.5f,-0.5f,-1,0,0,-0.5f,0.5f,0.5f,-1,0,0,-0.5f,-0.5f,0.5f,-1,0,0,
        0.5f,-0.5f,-0.5f,1,0,0,0.5f,0.5f,-0.5f,1,0,0,0.5f,0.5f,0.5f,1,0,0,0.5f,-0.5f,0.5f,1,0,0,
        -0.5f,-0.5f,-0.5f,0,-1,0,0.5f,-0.5f,-0.5f,0,-1,0,0.5f,-0.5f,0.5f,0,-1,0,-0.5f,-0.5f,0.5f,0,-1,0,
        -0.5f,0.5f,-0.5f,0,1,0,0.5f,0.5f,-0.5f,0,1,0,0.5f,0.5f,0.5f,0,1,0,-0.5f,0.5f,0.5f,0,1,0,
    };
    unsigned int idx[]={0,1,2,2,3,0,4,5,6,6,7,4,8,9,10,10,11,8,12,13,14,14,15,12,16,17,18,18,19,16,20,21,22,22,23,20};
    unsigned int VAO,VBO,EBO;
    glGenVertexArrays(1,&VAO);glGenBuffers(1,&VBO);glGenBuffers(1,&EBO);
    glBindVertexArray(VAO);
    glBindBuffer(GL_ARRAY_BUFFER,VBO);glBufferData(GL_ARRAY_BUFFER,sizeof(v),v,GL_STATIC_DRAW);
    glBindBuffer(GL_ELEMENT_ARRAY_BUFFER,EBO);glBufferData(GL_ELEMENT_ARRAY_BUFFER,sizeof(idx),idx,GL_STATIC_DRAW);
    glVertexAttribPointer(0,3,GL_FLOAT,GL_FALSE,6*sizeof(float),(void*)0);glEnableVertexAttribArray(0);
    glVertexAttribPointer(1,3,GL_FLOAT,GL_FALSE,6*sizeof(float),(void*)(3*sizeof(float)));glEnableVertexAttribArray(1);
    glBindVertexArray(0);return VAO;
}

unsigned int buildArrowVAO(int& cnt){
    std::vector<float> v;
    v.insert(v.end(),{0,0,0,0,0.65f,0});
    int seg=12;float r=0.04f,base=0.65f,tip=1.0f;
    for(int i=0;i<seg;i++){
        float a=2*3.14159f*i/seg,b=2*3.14159f*(i+1)/seg;
        v.insert(v.end(),{r*cos(a),base,r*sin(a),r*cos(b),base,r*sin(b),0,tip,0});
    }
    cnt=v.size()/3;
    unsigned int VAO,VBO;
    glGenVertexArrays(1,&VAO);glGenBuffers(1,&VBO);
    glBindVertexArray(VAO);
    glBindBuffer(GL_ARRAY_BUFFER,VBO);glBufferData(GL_ARRAY_BUFFER,v.size()*sizeof(float),v.data(),GL_STATIC_DRAW);
    glVertexAttribPointer(0,3,GL_FLOAT,GL_FALSE,3*sizeof(float),(void*)0);glEnableVertexAttribArray(0);
    glBindVertexArray(0);return VAO;
}

void drawMesh(const SceneObject& obj, unsigned int cubeVAO, VE::Mesh& sph, VE::Mesh& cyl, VE::Mesh& pyr){
    switch(obj.type){
        case PrimitiveType::Cube:     glBindVertexArray(cubeVAO);glDrawElements(GL_TRIANGLES,36,GL_UNSIGNED_INT,0);break;
        case PrimitiveType::Sphere:   glBindVertexArray(sph.VAO);glDrawElements(GL_TRIANGLES,sph.indexCount,GL_UNSIGNED_INT,0);break;
        case PrimitiveType::Cylinder: glBindVertexArray(cyl.VAO);glDrawElements(GL_TRIANGLES,cyl.indexCount,GL_UNSIGNED_INT,0);break;
        case PrimitiveType::Pyramid:  glBindVertexArray(pyr.VAO);glDrawElements(GL_TRIANGLES,pyr.indexCount,GL_UNSIGNED_INT,0);break;
        case PrimitiveType::Model3D:  if(obj.model&&obj.model->loaded) obj.model->Draw();break;
    }
}

void addObject(std::vector<SceneObject>& objects, PrimitiveType type, int& sel){
    const char* n[]={"Cube","Sphere","Cylinder","Pyramid","Model"};
    objects.push_back({std::string(n[(int)type])+"_"+std::to_string(objects.size()+1),
        glm::vec3(0,0.5f,0),glm::vec3(0),glm::vec3(1),glm::vec3(0.8f,0.6f,0.3f),type});
    sel=(int)objects.size()-1;
}

void loadModelToScene(std::vector<SceneObject>& objects, const std::string& path, int& sel){
    SceneObject obj;
    obj.name="Model_"+std::to_string(objects.size()+1);
    obj.pos=glm::vec3(0,0,0);obj.rot=glm::vec3(0);obj.scale=glm::vec3(1);
    obj.color=glm::vec3(0.8f,0.7f,0.6f);obj.type=PrimitiveType::Model3D;
    obj.modelPath=path;obj.model=std::make_shared<VE::Model>();
    if(obj.model->Load(path)){objects.push_back(obj);sel=(int)objects.size()-1;}
}

{

    VE::Shader shader(vertSrc,fragSrc);

    VE::Shader gridShader(gridVert,gridFrag);
    VE::Shader gizmoShader(gizmoVert,gizmoFrag);
    VE::Shader skyboxShader(skyboxVert,skyboxFrag);
    VE::Grid grid(20);

    std::string skyboxBase="C:\\Users\\loboy\\Downloads\\VisualEngine\\assets\\skybox\\";
    VE::Skybox skybox({

    });

    VE::Mesh sphere=VE::CreateSphere(),cylinder=VE::CreateCylinder(),pyramid=VE::CreatePyramid();

    unsigned int arrowVAO=buildArrowVAO(arrowCnt);

    std::vector<SceneObject> objects={
        {"Cube_1",   glm::vec3(0,0.5f,0),glm::vec3(0),glm::vec3(1),glm::vec3(0.4f,0.6f,0.9f),PrimitiveType::Cube},

    };
    int sel=0;
    bool showSkybox=true;

    // Script editor
    VE::ScriptEditor scriptEditor;

    // Project Panel
    std::string projectRoot="C:\\Users\\loboy\\Downloads\\VisualEngine\\project";
    VE::ProjectPanel projectPanel(projectRoot);
    projectPanel.onModelDropped=[&](const std::string& path){
        loadModelToScene(objects,path,sel);
    };

        scriptEditor.open(path);
    };

    glEnable(GL_DEPTH_TEST);
    glEnable(GL_STENCIL_TEST);

    while(!window->ShouldClose())
    {
        float now=glfwGetTime();deltaTime=now-lastFrame;lastFrame=now;

        if(!io.WantCaptureKeyboard){
            if(glfwGetKey(native,GLFW_KEY_W)==GLFW_PRESS) camera.ProcessKeyboard(0,deltaTime);
            if(glfwGetKey(native,GLFW_KEY_S)==GLFW_PRESS) camera.ProcessKeyboard(1,deltaTime);
            if(glfwGetKey(native,GLFW_KEY_A)==GLFW_PRESS) camera.ProcessKeyboard(2,deltaTime);

            auto& obj=objects[sel];
            scriptEditor.lua.objX=obj.pos.x;

            scriptEditor.lua.objZ=obj.pos.z;

            scriptEditor.lua.pullObjectData();

            obj.pos.y=scriptEditor.lua.objY;
            obj.pos.z=scriptEditor.lua.objZ;
        }

        glm::mat4 view=camera.GetViewMatrix();
        glm::mat4 proj=camera.GetProjectionMatrix(1280.0f/720.0f);

                }
                if(hitAxis!=GizmoAxis::None){
                    dragAxis=hitAxis;
                    dragStartPos=obj.pos;dragStartRot=obj.rot;dragStartScale=obj.scale;
                    glm::vec3 axDir=axes[(int)hitAxis-1];
                    glm::vec3 plN=glm::normalize(glm::cross(axDir,glm::cross(camera.Front,axDir)));
                    float t=rayPlaneT(ray,plN,obj.pos);
                    dragStartHit=t>0?ray.origin+ray.dir*t:obj.pos;
                } else {
                    float bestObj=1e9f;int bestI=-1;
                    for(int i=0;i<(int)objects.size();i++){

                        if(rayAABB(ray,objects[i].pos,objects[i].scale*0.55f,t)&&t<bestObj){bestObj=t;bestI=i;}

                    if(bestI>=0) sel=bestI;

                Ray ray=screenToRay(mouseX,mouseY,1280,720,view,proj);

                glm::vec3 axDir=axes[ai];

                float t=rayPlaneT(ray,plN,dragStartPos);
                if(t>0){
                    glm::vec3 hit=ray.origin+ray.dir*t;
                    float p=glm::dot(hit-dragStartHit,axDir);
                    if(gizmoMode==GizmoMode::Move) obj.pos=dragStartPos+axDir*p;

                }
            }

        leftClickThisFrame=false;

        glClearColor(0.08f,0.08f,0.12f,1);
        glClear(GL_COLOR_BUFFER_BIT|GL_DEPTH_BUFFER_BIT|GL_STENCIL_BUFFER_BIT);

        if(showSkybox) skybox.Draw(skyboxShader.ID,view,proj);
        grid.Draw(gridShader.ID,view,proj);

        glStencilOp(GL_KEEP,GL_KEEP,GL_REPLACE);
        shader.Use();
        glUniformMatrix4fv(glGetUniformLocation(shader.ID,"view"),      1,GL_FALSE,glm::value_ptr(view));
        glUniformMatrix4fv(glGetUniformLocation(shader.ID,"projection"),1,GL_FALSE,glm::value_ptr(proj));
        glUniform3f(glGetUniformLocation(shader.ID,"lightPos"),2,5,3);
        glUniform3f(glGetUniformLocation(shader.ID,"lightColor"),1,1,1);
        glUniform3f(glGetUniformLocation(shader.ID,"viewPos"),camera.Position.x,camera.Position.y,camera.Position.z);

        for(int i=0;i<(int)objects.size();i++){
            auto& obj=objects[i];
            if(i==sel){glStencilFunc(GL_ALWAYS,1,0xFF);glStencilMask(0xFF);}
            else{glStencilFunc(GL_ALWAYS,0,0xFF);glStencilMask(0x00);}
            glm::mat4 model=glm::mat4(1);
            model=glm::translate(model,obj.pos);
            model=glm::rotate(model,glm::radians(obj.rot.x),glm::vec3(1,0,0));
            model=glm::rotate(model,glm::radians(obj.rot.y),glm::vec3(0,1,0));
            model=glm::rotate(model,glm::radians(obj.rot.z),glm::vec3(0,0,1));

            model=glm::rotate(model,glm::radians(obj.rot.y),glm::vec3(0,1,0));
            model=glm::rotate(model,glm::radians(obj.rot.z),glm::vec3(0,0,1));

            glBindVertexArray(arrowVAO);
            glm::vec4 cols[3]={glm::vec4(1,0.15f,0.15f,1),glm::vec4(0.15f,1,0.15f,1),glm::vec4(0.15f,0.4f,1,1)};
            float rA[3]={-90,0,90};
            glm::vec3 rX[3]={glm::vec3(0,0,1),glm::vec3(0,1,0),glm::vec3(1,0,0)};
            for(int i=0;i<3;i++){
                glm::vec4 col=cols[i];
                if(dragAxis==(GizmoAxis)(i+1)) col=glm::vec4(1,1,0.2f,1);
                glm::mat4 m=glm::mat4(1);
                m=glm::translate(m,obj.pos);
                m=glm::rotate(m,glm::radians(rA[i]),rX[i]);
                m=glm::scale(m,glm::vec3(gs));
                glm::mat4 mvp=proj*view*m;
                glUniformMatrix4fv(glGetUniformLocation(gizmoShader.ID,"mvp"),1,GL_FALSE,glm::value_ptr(mvp));
                glUniform4f(glGetUniformLocation(gizmoShader.ID,"color"),col.r,col.g,col.b,col.a);
                glDrawArrays(GL_TRIANGLES,2,arrowCnt-2);
                glLineWidth(2.5f);
                glDrawArrays(GL_LINES,0,2);
            }
        }

        // ImGui
        ImGui_ImplOpenGL3_NewFrame();
        ImGui_ImplGlfw_NewFrame();
        ImGui::NewFrame();

        // Toolbar
        ImGui::SetNextWindowPos(ImVec2(240,0),ImGuiCond_Always);
        ImGui::SetNextWindowSize(ImVec2(800,42),ImGuiCond_Always);
        ImGui::Begin("##toolbar",nullptr,
            ImGuiWindowFlags_NoResize|ImGuiWindowFlags_NoMove|ImGuiWindowFlags_NoCollapse|
            ImGuiWindowFlags_NoTitleBar|ImGuiWindowFlags_NoScrollbar);
        ImGui::SetCursorPosY(8);
        bool mA=(gizmoMode==GizmoMode::Move);
        if(mA) ImGui::PushStyleColor(ImGuiCol_Button,ImVec4(0.3f,0.6f,1,1));
        if(ImGui::Button("  Move (G)  ")) gizmoMode=GizmoMode::Move;
        if(mA) ImGui::PopStyleColor();
        ImGui::SameLine();
        bool rA=(gizmoMode==GizmoMode::Rotate);
        if(rA) ImGui::PushStyleColor(ImGuiCol_Button,ImVec4(0.3f,0.6f,1,1));
        if(ImGui::Button(" Rotate (R) ")) gizmoMode=GizmoMode::Rotate;
        if(rA) ImGui::PopStyleColor();
        ImGui::SameLine();
        bool sA=(gizmoMode==GizmoMode::Scale);
        if(sA) ImGui::PushStyleColor(ImGuiCol_Button,ImVec4(0.3f,0.6f,1,1));
        if(ImGui::Button("  Scale (T)  ")) gizmoMode=GizmoMode::Scale;
        if(sA) ImGui::PopStyleColor();
        ImGui::SameLine(0,20);
        ImGui::Checkbox("Skybox",&showSkybox);
        ImGui::SameLine(0,20);
        ImGui::TextColored(ImVec4(0.5f,0.5f,0.6f,1),"W/A/S/D  |  RMB rotate  |  LMB select/drag");
        ImGui::End();

        // Scene Hierarchy
        ImGui::SetNextWindowPos(ImVec2(0,0),ImGuiCond_Always);
        ImGui::SetNextWindowSize(ImVec2(240,440),ImGuiCond_Always);
        ImGui::Begin("Scene Hierarchy",nullptr,
            ImGuiWindowFlags_NoResize|ImGuiWindowFlags_NoMove|ImGuiWindowFlags_NoCollapse);
        ImGui::TextColored(ImVec4(0.5f,0.75f,1,1),"Objects (%d)",(int)objects.size());
        ImGui::Separator();
        const char* icons[]={"[Cube]","[Sphere]","[Cylinder]","[Pyramid]","[Model]"};
        for(int i=0;i<(int)objects.size();i++){
            std::string lbl=std::string(icons[(int)objects[i].type])+" "+objects[i].name;
            if(ImGui::Selectable(lbl.c_str(),i==sel)) sel=i;
        }
        ImGui::Separator();
        ImGui::TextColored(ImVec4(0.5f,0.75f,1,1),"Add Primitive");
        if(ImGui::Button("+ Cube",    ImVec2(-1,0))) addObject(objects,PrimitiveType::Cube,    sel);
        if(ImGui::Button("+ Sphere",  ImVec2(-1,0))) addObject(objects,PrimitiveType::Sphere,  sel);
        if(ImGui::Button("+ Cylinder",ImVec2(-1,0))) addObject(objects,PrimitiveType::Cylinder,sel);
        if(ImGui::Button("+ Pyramid", ImVec2(-1,0))) addObject(objects,PrimitiveType::Pyramid, sel);
        ImGui::Separator();
        if(!objects.empty()&&ImGui::Button("- Remove Selected",ImVec2(-1,0))){
            objects.erase(objects.begin()+sel);
            if(sel>=(int)objects.size()) sel=(int)objects.size()-1;
        }
        ImGui::End();

        // Project Panel
        projectPanel.Draw();

        // Inspector
        ImGui::SetNextWindowPos(ImVec2(1040,0),ImGuiCond_Always);
        ImGui::SetNextWindowSize(ImVec2(240,720),ImGuiCond_Always);
        ImGui::Begin("Inspector",nullptr,
            ImGuiWindowFlags_NoResize|ImGuiWindowFlags_NoMove|ImGuiWindowFlags_NoCollapse);
        if(!objects.empty()&&sel>=0&&sel<(int)objects.size()){
            auto& obj=objects[sel];
            ImGui::TextColored(ImVec4(0.5f,0.75f,1,1),"%s",obj.name.c_str());
            ImGui::TextDisabled("%s",icons[(int)obj.type]);
            if(obj.type==PrimitiveType::Model3D&&obj.model)
                ImGui::TextDisabled("Verts: %d",obj.model->vertexCount);
            ImGui::Separator();
            ImGui::Text("Transform");
            ImGui::DragFloat3("Position",glm::value_ptr(obj.pos),  0.05f);
            ImGui::DragFloat3("Rotation",glm::value_ptr(obj.rot),  0.5f);
            ImGui::DragFloat3("Scale",   glm::value_ptr(obj.scale),0.05f,0.01f,20.0f);
            ImGui::Separator();
            ImGui::Text("Material");
            ImGui::ColorEdit3("Color",glm::value_ptr(obj.color));
            ImGui::Separator();
            ImGui::Text("Script");
            if(ImGui::Button("New Script & Attach",ImVec2(-1,0))){
                std::string scriptPath=projectRoot+"\\Assets\\Scripts\\"+obj.name+".lua";
                obj.scriptPath=scriptPath;
                scriptEditor.open(scriptPath);
            }
            if(!obj.scriptPath.empty()){
                ImGui::TextDisabled("%s",obj.scriptPath.c_str());
                if(ImGui::Button("Edit Script",ImVec2(-1,0)))
                    scriptEditor.open(obj.scriptPath);
            }
        } else {
            ImGui::TextDisabled("No object selected");
        }
        ImGui::Separator();
        ImGui::TextColored(ImVec4(0.5f,1,0.5f,1),"FPS: %.0f",io.Framerate);
        ImGui::End();

        // Script Editor window
        scriptEditor.draw();

        ImGui::Render();
        ImGui_ImplOpenGL3_RenderDrawData(ImGui::GetDrawData());
        window->OnUpdate();
    }

    ImGui_ImplOpenGL3_Shutdown();ImGui_ImplGlfw_Shutdown();ImGui::DestroyContext();
    shader.Delete();outlineShader.Delete();gridShader.Delete();gizmoShader.Delete();skyboxShader.Delete();
    delete window;return 0;
}
